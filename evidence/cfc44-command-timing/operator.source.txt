import fs from 'node:fs';import {spawn} from 'node:child_process';import {DatabaseSync} from 'node:sqlite';import {performance} from 'node:perf_hooks';
export const m=JSON.parse(fs.readFileSync('/srv/dev/work/orchestrator/raf-develop/evidence/cfc44-command-timing/stand.json'));
export const ev='/srv/dev/work/orchestrator/raf-develop/evidence/cfc44-command-timing';
export const environment=Object.fromEntries(Object.entries(process.env).filter(([k])=>!k.startsWith('RAF_')&&!['TMUX','TMUX_PANE'].includes(k)));
Object.assign(environment,{HOME:m.home,RAF_QA_HOME:m.home,RAF_ORCHESTRATOR_ROOT:m.root,TMPDIR:m.base+'/tmp'});
export function snapshot(){if(!fs.existsSync(m.statePath))return null;const db=new DatabaseSync(m.statePath,{readOnly:true});try{return {timings:db.prepare('SELECT * FROM command_timings ORDER BY command').all(),audit:db.prepare('SELECT * FROM qa_timing_audit ORDER BY id').all()};}finally{db.close();}}
export async function cli(label,args,overrides={}){const before=snapshot(),start=performance.now(),startedAt=new Date().toISOString();const command=[process.execPath,m.code+'/bin/raf.ts',...args];const child=spawn(command[0],command.slice(1),{cwd:m.code,env:{...environment,...overrides},stdio:['ignore','pipe','pipe']});let stdout='',stderr='';child.stdout.on('data',c=>stdout+=c);child.stderr.on('data',c=>stderr+=c);const result=await new Promise(resolve=>child.once('close',(code,signal)=>resolve({cliExitCode:code,signal})));const wallMs=performance.now()-start;fs.writeFileSync(ev+'/'+label+'.stdout',stdout);fs.writeFileSync(ev+'/'+label+'.stderr',stderr);const data={command,environment:{HOME:overrides.HOME??environment.HOME,RAF_QA_HOME:overrides.RAF_QA_HOME??environment.RAF_QA_HOME,RAF_ORCHESTRATOR_ROOT:environment.RAF_ORCHESTRATOR_ROOT,TMPDIR:environment.TMPDIR},startedAt,completedAt:new Date().toISOString(),wallMs,...result,childExitCode:null,childNote:'Direct CLI operation; no separate payload process',before,after:snapshot()};fs.writeFileSync(ev+'/'+label+'.json',JSON.stringify(data,null,2)+'\n');return {...data,stdout,stderr};}
